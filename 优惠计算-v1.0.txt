      <b>统考：</b>公共课 + 统考专业课合并计科数，1 科 85 折、2 科 8 折、3 科及以上 75 折（封顶），折扣后再减 200 元。<br>
      <b>自命题：</b>公共课单独按科数享 85 / 8 / 75 折并减 200 元；自命题专业课按课时计价（C9 700 元、普通 985 400 元、211 及以下 350 元），20 课时及以上 C9 92 折、其他 9 折；40 课时及以上 C9 9 折、其他 85 折，<b>自命题部分不参与公共课折扣体系，也不参与 200 元减免</b>。

<script>
(function () {
  /* ============================================================
     常量
     ============================================================ */
  const PRICE_MAJOR   = 7980;
  const PRICE_GENERAL = 6980;
  const CUT           = 200;
  const MAX           = 99;
  const MIN_RATE      = 0.75;
  const HOUR_RATES = { c9: 700, '985': 400, '211': 350 };

  const state = {
    generalCount: 0,
    majorCount: 0,
    majorMode: 'tongkao',
    schoolLevel: 'c9',
    hours: 0
  };

  const $ = id => document.getElementById(id);

  const el = {
    generalInput: $('generalInput'),
    generalMinus: $('generalMinus'),
    generalPlus: $('generalPlus'),
    majorInput: $('majorInput'),
    majorMinus: $('majorMinus'),
    majorPlus: $('majorPlus'),
    hourRateHint: $('hourRateHint'),
    hoursInput: $('hoursInput'),
    panelTongkao: $('panelTongkao'),
    panelZpm: $('panelZpm'),
    finalPrice: $('finalPrice'),
    rateTag: $('rateTag'),
    detail: $('detail'),
    lineSubtotal: $('lineSubtotal'),
    lineDiscountLabel: $('lineDiscountLabel'),

    lineCutWrap: $('lineCutWrap'),
    calcLine: $('calcLine'),
    copyCalc: $('copyCalc')
  };

  /* ============================================================
     工具函数
     ============================================================ */
  function money(n) {
    return '¥' + Number(n).toLocaleString('zh-CN', { maximumFractionDigits: 2 });
  }

  // 统考折扣体系（也用于自命题模式下公共课单独折扣）
  function getTongkaoRate(count) {
    let rate = 1, text = '请选择课程数量';
    if (count >= 3)       { rate = 0.75; text = '3科及以上 75折'; }
    else if (count === 2) { rate = 0.80; text = '2科 8折'; }
    else if (count === 1) { rate = 0.85; text = '1科 85折'; }
    if (rate < MIN_RATE) { rate = MIN_RATE; text = '3科及以上 75折（已封顶）'; }
    return { rate: rate, text: text };
  }

  // 自命题折扣
  function getZpmRate(hours, level) {
    if (hours >= 40) {
      return {
        rate: level === 'c9' ? 0.90 : 0.85,
        text: level === 'c9' ? '40课时及以上 9折' : '40课时及以上 85折'
      };
    }
    if (hours >= 20) {
      return {
        rate: level === 'c9' ? 0.92 : 0.90,
        text: level === 'c9' ? '20课时及以上 92折' : '20课时及以上 9折'
      };
    }
    return { rate: 1, text: '无折扣' };
  }

  /* --- 统考公式 --- */
  function buildFormulaTongkao(g, m, rate, finalPrice) {
    const parts = [];
    if (g > 0) parts.push(PRICE_GENERAL + '×' + g);
    if (m > 0) parts.push(PRICE_MAJOR + '×' + m);
    if (parts.length === 0) return '';

    let expr = parts.length > 1 ? '(' + parts.join('+') + ')' : '(' + parts[0] + ')';
    return expr + '×' + rate.toFixed(2) + '−' + CUT + '=' + finalPrice;
  }

  /* --- 自命题公式：公共课与自命题各自独立计算 --- */
  function buildFormulaZpm(g, gRate, hours, hourRate, zRate, finalPrice) {
    const parts = [];

    // 公共课部分：(6980×2)×0.85−200
    if (g > 0) {
      parts.push('(' + PRICE_GENERAL + '×' + g + ')×' + gRate.toFixed(2) + '−' + CUT);
    }

    // 自命题部分：(30×400×0.90) 或 (30×400)
    if (hours > 0) {
      let s = '(' + hours + '×' + hourRate;
      if (zRate < 1) s += '×' + zRate.toFixed(2);
      s += ')';
      parts.push(s);
    }

    if (parts.length === 0) return '';
    return parts.join('+') + '=' + finalPrice;
  }

  /* ============================================================
     核心：刷新界面
     ============================================================ */
  function update() {
    const g = state.generalCount;

    el.generalInput.value = g;
    el.generalMinus.disabled = g <= 0;
    el.generalPlus.disabled = g >= MAX;

    el.majorInput.value = state.majorCount;
    el.majorMinus.disabled = state.majorCount <= 0;
    el.majorPlus.disabled = state.majorCount >= MAX;

    let subtotal = 0;
    let discountAmount = 0;
    let final = 0;
    let formula = '';
    let showCut = false;
    let discountLabelText = '折扣优惠';
    let tagText = '请选择课程数量';

    if (state.majorMode === 'tongkao') {
      /* ---------- 统考模式：公共课 + 统考专业课合并折扣 ---------- */
      const m = state.majorCount;
      const count = g + m;
      subtotal = g * PRICE_GENERAL + m * PRICE_MAJOR;

      const r = getTongkaoRate(count);
      const rate = r.rate;
      discountAmount = subtotal * (1 - rate);
      final = count > 0 ? Math.max(subtotal * rate - CUT, 0) : 0;
      formula = buildFormulaTongkao(g, m, rate, final);
      showCut = true;

      if (count > 0) {
        tagText = r.text + '，再减 ¥200';
        if (rate < 1) discountLabelText = '折扣优惠（' + r.text + '）';
      } else {
        tagText = '请选择课程数量';
      }
    } else {
      /* ---------- 自命题模式 ----------
         公共课：单独按科数享 85/8/75 折，并减 200
         自命题：按课时独立折扣，不参与公共课折扣体系与 200 减免
      ------------------------------------------- */
      const hours = state.hours;
      const level = state.schoolLevel;
      const hourRate = HOUR_RATES[level];

      const gRateObj = getTongkaoRate(g);
      const gRate = g > 0 ? gRateObj.rate : 1;

      const zRateObj = getZpmRate(hours, level);
      const zRate = zRateObj.rate;

      const gRaw = g * PRICE_GENERAL;
      const zRaw = hours * hourRate;
      subtotal = gRaw + zRaw;

      const gDiscount = gRaw * (1 - gRate);
      const zDiscount = zRaw * (1 - zRate);
      discountAmount = gDiscount + zDiscount;

      const gFinal = g > 0 ? Math.max(gRaw * gRate - CUT, 0) : 0;
      const zFinal = zRaw * zRate;
      final = gFinal + zFinal;

      formula = buildFormulaZpm(g, gRate, hours, hourRate, zRate, final);
      showCut = g > 0;

      // 折扣标签
      const dLabels = [];
      if (g > 0 && gRate < 1) dLabels.push('公共课 ' + gRateObj.text);
      if (hours > 0 && zRate < 1) dLabels.push('自命题 ' + zRateObj.text);
      if (dLabels.length > 0) discountLabelText = '折扣优惠（' + dLabels.join('·') + '）';

      // 结果卡片标签
      const tParts = [];
      if (g > 0) {
        tParts.push('公共课 ' + gRateObj.text + '，再减 ¥200');
      }
      if (hours > 0) {
        tParts.push(zRate < 1 ? '自命题 ' + zRateObj.text : '自命题 无折扣');
      }
      if (tParts.length === 0) {
        tagText = '请填写课时数或选择公共课';
      } else {
        tagText = tParts.join(' · ');
      }
    }

    el.finalPrice.textContent = money(final);
    el.rateTag.textContent = tagText;

    if (subtotal > 0) {
      el.detail.classList.remove('hidden');
      el.lineSubtotal.textContent = money(subtotal);
      el.lineDiscountLabel.textContent = discountLabelText;
      el.lineDiscount.textContent = '-' + money(discountAmount);
      el.lineCutWrap.style.display = showCut ? '' : 'none';
      el.calcLine.textContent = formula;
    } else {
      el.detail.classList.add('hidden');
      el.calcLine.textContent = '';
    }

    if (copyTimer) { clearTimeout(copyTimer); copyTimer = null; }
    if (el.copyCalc) {
      el.copyCalc.classList.remove('copied');
      el.copyCalc.title = '复制计算过程';
    }
  }

  /* ============================================================
     自定义下拉组件
     ============================================================ */
  function initDropdown(rootId, onChange) {
    const root = document.getElementById(rootId);
    if (!root) return;

    const trigger = root.querySelector('.gt-dropdown-trigger');
    const label = root.querySelector('.gt-dropdown-label');
    const items = root.querySelectorAll('.gt-dropdown-item');

    const current = root.dataset.value;
    items.forEach(function (item) {
      if (item.dataset.value === current) {
        item.classList.add('selected');
        const s = item.querySelector('span');
        if (s) label.textContent = s.textContent;
      }
    });

    trigger.addEventListener('click', function (e) {
      e.stopPropagation();
      document.querySelectorAll('.gt-dropdown.open').forEach(function (d) {
        if (d !== root) d.classList.remove('open');
      });
      root.classList.toggle('open');
    });

    items.forEach(function (item) {
      item.addEventListener('click', function (e) {
        e.stopPropagation();

        const value = item.dataset.value;
        const s = item.querySelector('span');
        const text = s ? s.textContent : item.textContent.trim();

        items.forEach(function (it) { it.classList.remove('selected'); });
        item.classList.add('selected');

        root.dataset.value = value;
        label.textContent = text;
        root.classList.remove('open');

        if (onChange) onChange(value, text);
      });
    });
  }

  document.addEventListener('click', function () {
    document.querySelectorAll('.gt-dropdown.open').forEach(function (d) {
      d.classList.remove('open');
    });
  });

  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape') {
      document.querySelectorAll('.gt-dropdown.open').forEach(function (d) {
        d.classList.remove('open');
      });
    }
  });

  initDropdown('dropdownMajorMode', function (value) {
    state.majorMode = value;
    if (value === 'tongkao') {
      el.panelTongkao.classList.remove('hidden');
      el.panelZpm.classList.add('hidden');
    } else {
      el.panelTongkao.classList.add('hidden');
      el.panelZpm.classList.remove('hidden');
    }
    update();
  });

  initDropdown('dropdownSchoolLevel', function (value) {
    state.schoolLevel = value;
    el.hourRateHint.textContent = HOUR_RATES[value] + ' 元 / 课时';
    update();
  });

  /* ============================================================
     事件绑定：公共课 / 统考数量
     ============================================================ */
  el.generalMinus.addEventListener('click', function () {
    if (state.generalCount > 0) { state.generalCount--; update(); }
  });
  el.generalPlus.addEventListener('click', function () {
    if (state.generalCount < MAX) { state.generalCount++; update(); }
  });

  el.majorMinus.addEventListener('click', function () {
    if (state.majorCount > 0) { state.majorCount--; update(); }
  });
  el.majorPlus.addEventListener('click', function () {
    if (state.majorCount < MAX) { state.majorCount++; update(); }
  });

  /* ============================================================
     事件绑定：课时数
     ============================================================ */
  el.hoursInput.addEventListener('input', function () {
    const raw = el.hoursInput.value.trim();
    if (raw === '') {
      state.hours = 0;
    } else {
      const v = parseFloat(raw);
      state.hours = (isNaN(v) || v < 0) ? 0 : Math.min(Math.floor(v), 9999);
    }
    update();
  });

  el.hoursInput.addEventListener('blur', function () {
    el.hoursInput.value = state.hours;
  });

  /* ============================================================
     计算过程一键复制
     ============================================================ */
  let copyTimer = null;

  function fallbackCopy(text) {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.setAttribute('readonly', '');
    ta.style.position = 'fixed';
    ta.style.top = '0';
    ta.style.left = '-9999px';
    document.body.appendChild(ta);
    ta.focus();
    ta.select();
    ta.setSelectionRange(0, ta.value.length);
    let ok = false;
    try { ok = document.execCommand('copy'); } catch (e) { ok = false; }
    document.body.removeChild(ta);
    return ok;
  }

  function copyToClipboard(text) {
    if (navigator.clipboard && window.isSecureContext) {
      return navigator.clipboard.writeText(text)
        .then(function () { return true; })
        .catch(function () { return fallbackCopy(text); });
    }
    return Promise.resolve(fallbackCopy(text));
  }

  el.copyCalc.addEventListener('click', function () {
    const text = el.calcLine.textContent.trim();
    if (!text) return;

    copyToClipboard(text).then(function (ok) {
      if (!ok) return;
      el.copyCalc.classList.add('copied');
      el.copyCalc.title = '已复制';
      if (copyTimer) clearTimeout(copyTimer);
      copyTimer = setTimeout(function () {
        el.copyCalc.classList.remove('copied');
        el.copyCalc.title = '复制计算过程';
        copyTimer = null;
      }, 1600);
    });
  });

  /* ============================================================
     粘贴求和（完全独立）
     ============================================================ */
  const pasteArea    = $('pasteArea');
  const pasteTip     = $('pasteTip');
  const clearPaste   = $('clearPaste');
  const sumValue     = $('sumValue');
  const sumMeta      = $('sumMeta');
  const sumBreakdown = $('sumBreakdown');

  const NUMBER_RE = /-?\d[\d,]*(?:\.\d+)?/g;

  function normalizeText(text) {
    return text
      .replace(/[０-９]/g, function (ch) {
        return String.fromCharCode(ch.charCodeAt(0) - 0xFEE0);
      })
      .replace(/[，]/g, ',')
      .replace(/[。．]/g, '.')
      .replace(/[－—–]/g, '-')
      .replace(/[＋]/g, '+')
      .replace(/[\u00A0\u202F\u2007\u2060]/g, ' ')
      .replace(/[\t\v\f\u0007\u001C-\u001F\u2028\u2029]+/g, '\n')
      .replace(/\r\n?/g, '\n');
  }

  function parseNumbers(text) {
    const normalized = normalizeText(text);
    const lines = normalized.split('\n');
    const values = [];
    let skippedLines = 0;

    lines.forEach(function (line) {
      const t = line.trim();
      if (!t) return;

      const matches = t.match(NUMBER_RE);
      if (matches && matches.length > 0) {
        matches.forEach(function (m) {
          const v = parseFloat(m.replace(/,/g, ''));
          if (!isNaN(v)) values.push(v);
        });
      } else {
        skippedLines++;
      }
    });

    return { values: values, skippedLines: skippedLines };
  }

  function formatSum(n) {
    return '¥' + Number(n).toLocaleString('zh-CN', { maximumFractionDigits: 2 });
  }

  function setTip(text, type) {
    pasteTip.textContent = text;
    pasteTip.className = 'paste-tip' + (type ? ' ' + type : '');
  }

  function refreshSum() {
    const raw = pasteArea.value;

    if (!raw.trim()) {
      sumValue.textContent = '¥0';
      sumMeta.textContent = '尚未粘贴数据';
      sumBreakdown.classList.add('hidden');
      sumBreakdown.textContent = '';
      setTip('等待粘贴数据…');
      return;
    }

    const result = parseNumbers(raw);
    const values = result.values;
    const skippedLines = result.skippedLines;

    if (values.length === 0) {
      sumValue.textContent = '¥0';
      sumMeta.textContent = '未识别到数字';
      sumBreakdown.classList.add('hidden');
      sumBreakdown.textContent = '';
      setTip('未识别到数字，请检查粘贴内容', 'err');
      return;
    }

    let sum = 0;
    for (let i = 0; i < values.length; i++) sum += values[i];

    sumValue.textContent = formatSum(sum);

    let meta = '共 ' + values.length + ' 个数字';
    if (skippedLines > 0) meta += '，忽略 ' + skippedLines + ' 行无数字内容';
    sumMeta.textContent = meta;

    if (values.length > 1) {
      sumBreakdown.textContent = values.join(' + ') + ' = ' + sum;
      sumBreakdown.classList.remove('hidden');
    } else {
      sumBreakdown.classList.add('hidden');
      sumBreakdown.textContent = '';
    }

    let tip = '已识别 ' + values.length + ' 个数字，合计 ' + formatSum(sum);
    let type = 'ok';
    if (skippedLines > 0) {
      tip += '（忽略 ' + skippedLines + ' 行）';
      type = 'warn';
    }
    setTip(tip, type);
  }

  pasteArea.addEventListener('paste', function (e) {
    const clipboard = e.clipboardData || window.clipboardData;
    if (!clipboard) return;

    let text = clipboard.getData('text/plain');
    if (!text) {
      const html = clipboard.getData('text/html');
      if (html) {
        text = html
          .replace(/<br\s*\/?>/gi, '\n')
          .replace(/<\/(p|div|tr|li|h[1-6])>/gi, '\n')
          .replace(/<[^>]+>/g, '')
          .replace(/&nbsp;/gi, ' ')
          .replace(/&amp;/gi, '&')
          .replace(/&lt;/gi, '<')
          .replace(/&gt;/gi, '>');
      }
    }
    if (!text) return;

    e.preventDefault();

    const start = pasteArea.selectionStart;
    const end = pasteArea.selectionEnd;
    const before = pasteArea.value.slice(0, start);
    const after = pasteArea.value.slice(end);
    pasteArea.value = before + text + after;

    const caret = start + text.length;
    pasteArea.selectionStart = pasteArea.selectionEnd = caret;

    refreshSum();
  });

  pasteArea.addEventListener('input', refreshSum);

  clearPaste.addEventListener('click', function () {
    pasteArea.value = '';
    refreshSum();
    pasteArea.focus();
  });

  /* ============================================================
     初始化
     ============================================================ */
  el.hourRateHint.textContent = HOUR_RATES[state.schoolLevel] + ' 元 / 课时';
  refreshSum();
  update();
})();
